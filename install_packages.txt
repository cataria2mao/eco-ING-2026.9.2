# ---------- 1. Web 服务（web_server.py） ----------
pip install fastapi
pip install "uvicorn[standard]"
pip install websockets

# ---------- 2. LangGraph / LangChain 框架（agent_core.py, subgraph_agent.py） ----------
pip install "langgraph>=1.0"
pip install "langchain-core>=1.0"
pip install "langchain-deepseek>=1.0"
pip install langchain-text-splitters

# ---------- 3. 向量数据库（build_vector_db.py, agent_core.py） ----------
pip install "chromadb>=1.0"

# ---------- 4. 数据处理与文件读写（PGstore.py, subgraph_agent.py） ----------
pip install "pandas>=2.0"
pip install "openpyxl>=3.0"
pip install "python-docx>=1.0"
pip install sqlalchemy
pip install psycopg2-binary

# ---------- 5. 环境变量与网络请求（agent_core.py, build_vector_db.py） ----------
pip install "python-dotenv>=1.0"
pip install "requests>=2.0"

# ---------- 6. 重排序模型 BGE-Reranker（agent_core.py） ----------
pip install "FlagEmbedding>=1.3.0"
pip install huggingface_hub
pip install torch

# 说明：FlagEmbedding 会自动安装 transformers、sentence-transformers 等依赖；
# 如需 GPU 版 PyTorch，请按显卡 CUDA 版本到 https://pytorch.org 选择对应安装命令，
# 例如（CUDA 12.1）：
# pip install torch --index-url https://download.pytorch.org/whl/cu121
